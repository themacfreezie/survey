Next I extract from the MARSS object state estimates and confidence intervals, as well as Y_t predictions and those intervals as well. This is where I’m concerned I might not be grabbing the right information from the fitted model, specifically the “ytT” bit. If this looks correct to you than these plots should be good.

# pull fitted states from MARSS object
states_coho <- tsSmooth(ssm_coho, type = "xtT")

predobs_coho <- predict(ssm_coho, type = "ytT", interval = "prediction", level = 0.95)
obs_coho <- predobs_coho$pred

Now I’ve got to match the state key to the states in states_coho to assign each the proper popID.

names(states_coho)[names(states_coho) == ".rownames"] <- "state"
states_coho$state <- as.numeric(sub("^X", "", states_coho$state))
states_coho <- states_coho %>%
  left_join(keyX_coho, by = c("state" = "State")) %>%
  select(-state)
states_coho$Year <- states_coho$t + 1979
states_coho <- rename(states_coho, Value = .estimate, SE = .se)
states_coho <- states_coho[-c(1)]
states_coho$Dataset <- "fitted"

states_coho$upper95 <- states_coho$Value + (states_coho$SE * 1.96)
states_coho$lower95 <- states_coho$Value - (states_coho$SE * 1.96)

states_coho <- states_coho %>%
  mutate(PopID = as.character(PopID)) %>%
  left_join(
    pop_names %>% mutate(PopID = as.character(PopID)),
    by = "PopID"
  )

Next I bind these two datasets together (observations and underlying states). A step here drops all fitted state estimates from before or after the first and last observed year.

# row bind
plotdata_coho <- rbind(states_coho, obs_coho)

# drop estimates from before any observations were made
plotdata_coho <- plotdata_coho %>%
  group_by(PopID) %>%
  filter(
    trimws(Dataset) == "observed" | 
      (trimws(Dataset) == "fitted" & Year >= min(Year[trimws(Dataset) == "observed" & !is.na(Value)], na.rm = TRUE) &
         Year <= max(Year[trimws(Dataset) == "observed" & !is.na(Value)], na.rm = TRUE))) %>%
  ungroup()

Now that I have a dataset with both observations and fitted state estiamtes, I make my plot, including confidence intervals around each.

statecompare_cohoOBVAR <- ggplot(plotdata_coho, aes(x = Year, y = Value, color = Dataset, fill = Dataset)) +
  geom_ribbon(aes(ymin = lower95, ymax = upper95), alpha = 0.2, color = NA, show.legend = FALSE) +
  geom_line(linewidth = 0.8) +
  facet_wrap(~ COMMONPOPNAME, scales = "free_y") + 
  theme_minimal() +
  scale_color_manual(
    values = c("fitted" = "#D55E00", "observed" = "#0072B2"), 
    labels = c("fitted" = "State Estimate", "observed" = "Observation")
  ) +
  labs(
    title = "Coho population time series comparison by ESU (1980-2024)",
    x = "",
    y = "ln(NOSA)",
    color = ""
  ) +
  theme(
    legend.position = "bottom",
